MARKET DATA & QUALITY # ============================================================ @dataclass class MarketSnapshot: symbol: str timeframe: str bid: float ask: float timestamp: str open: float high: float low: float close: float source: str = "UNKNOWN" @property def spread_raw(self) -> float: return self.ask - self.bid @property def spread(self) -> float: return round(self.spread_raw, 5) @dataclass class DataQuality: source: str timestamp: str freshness_seconds: float reliability: float is_fact: bool = True conflict: bool = False status: str = "VALID" # ============================================================ # 3. SYSTEM PRE-FLIGHT SCAN # ============================================================ class PreFlight: REQUIRED_MODULES = ["DATA", "MAP", "PLAN", "NAVI"] MAX_SPREAD = {"XAUUSD": 0.8, "EURUSD": 0.00020, "DEFAULT": 0.002} MAX_FRESHNESS_SEC = 30 @staticmethod def run(snapshot: MarketSnapshot, event: Optional[EventState] = None, data_quality: Optional[DataQuality] = None) -> Dict[str, Any]: errors = [] if not snapshot.symbol: errors.append("MISSING_SYMBOL") if not snapshot.timeframe: errors.append("MISSING_TIMEFRAME") if snapshot.bid <= 0: errors.append("INVALID_BID") if snapshot.ask <= 0: errors.append("INVALID_ASK") if snapshot.ask < snapshot.bid: errors.append("ASK_BELOW_BID") if snapshot.high < snapshot.low: errors.append("INVALID_CANDLE_RANGE") max_spread = PreFlight.MAX_SPREAD.get(snapshot.symbol, PreFlight.MAX_SPREAD["DEFAULT"]) if snapshot.spread_raw > max_spread: errors.append(f"SPREAD_TOO_WIDE_{snapshot.spread_raw:.5f}_MAX_{max_spread}") if data_quality and data_quality.freshness_seconds > PreFlight.MAX_FRESHNESS_SEC: errors.append(f"STALE_DATA_{data_quality.freshness_seconds:.1f}s") if event and event.importance == "HIGH" and event.minutes_to_event is not None: if abs(event.minutes_to_event) < 15: errors.append(f"HIGH_IMPACT_NEWS_BLOCK_{event.name}_{event.minutes_to_event}min") if errors: return {"status": SystemStatus.HALT.value, "errors": errors, "spread": snapshot.spread, "modules": PreFlight.REQUIRED_MODULES} return {"status": SystemStatus.READY.value, "errors": [], "spread": snapshot.spread, "modules": PreFlight.REQUIRED_MODULES} # ============================================================ # 7. DATA VALIDATOR & QUALITY GATE # ============================================================ class DataValidator: @staticmethod def validate(snapshot: MarketSnapshot) -> Dict[str, Any]: errors = [] if snapshot.high < snapshot.low: errors.append("HIGH_LOWER_THAN_LOW") if not (snapshot.low <= snapshot.open <= snapshot.high): errors.append("OPEN_OUTSIDE_CANDLE") if not (snapshot.low <= snapshot.close <= snapshot.high): errors.append("CLOSE_OUTSIDE_CANDLE") if snapshot.bid <= 0 or snapshot.ask <= 0: errors.append("INVALID_BID_ASK") return {"valid": len(errors) == 0, "errors": errors} @staticmethod def quality_check(snapshot: MarketSnapshot) -> DataQuality: now = datetime.now(timezone.utc) try: snap_time = datetime.fromisoformat(snapshot.timestamp.replace('Z', '+00:00')) freshness = (now - snap_time).total_seconds() except Exception: freshness = 0.0 reliability = 1.0 if freshness > 15: reliability -= 0.3 if snapshot.spread_raw <= 0: reliability = 0.0 return DataQuality( source=snapshot.source, timestamp=snapshot.timestamp, freshness_seconds=max(0.0, freshness), reliability=max(0.0, reliability), status="VALID" if reliability >= 0.7 else "DEGRADED" ) # ============================================================ # 8. MAP ENGINE (INDICATOR-FREE & ATR DRIVEN) # ============================================================ class MAPEngine: def calc_atr(self, history: List[Dict], period: int = 14) -> float: if not history or len(history) < period + 1: return 0.0 trs = [] for i in range(1, len(history)): h, l, pc = history[i]['high'], history[i]['low'], history[i-1]['close'] tr = max(h - l, abs(h - pc), abs(l - pc)) trs.append(tr) return sum(trs[-period:]) / period if trs else 0.0 def build(self, snapshot: MarketSnapshot, history: Optional[List[Dict]] = None) -> MAPPicture: structure = self.detect_structure(snapshot, history) liquidity = self.detect_liquidity(snapshot, structure) session = self.detect_session() event = EventState(name=None, event_time=None, importance="NONE", minutes_to_event=None) intelligence = MarketIntelligence() location = self.detect_location(snapshot, structure) cycle = self.detect_market_cycle(structure) data_quality = DataValidator.quality_check(snapshot) picture = MAPPicture( snapshot=snapshot, structure=structure, liquidity=liquidity, session=session, event=event, intelligence=intelligence, location=location, market_cycle=cycle, data_quality=data_quality ) picture.quality_score = self.quality_gate(picture) picture.ready = picture.quality_score >= 70 and data_quality.reliability >= 0.7 if not picture.ready: picture.conflicts.append(f"QUALITY_GATE_{picture.quality_score}_RELIABILITY_{data_quality.reliability}") return picture def detect_structure(self, snapshot: MarketSnapshot, history: Optional[List[Dict]] = None) -> Structure: atr = self.calc_atr(history) if history else (snapshot.high - snapshot.low) if history and len(history) >= 20: highs = [c['high'] for c in history] lows = [c['low'] for c in history] closes = [c['close'] for c in history] swing_high = max(highs[-20:]) swing_low = min(lows[-20:]) if closes[-1] > max(highs[:-1]): trend, bos = "BULLISH", "BOS_UP" elif closes[-1] < min(lows[:-1]): trend, bos = "BEARISH", "BOS_DOWN" else: trend, bos = "NEUTRAL", None else: swing_high = snapshot.high swing_low = snapshot.low trend = "BULLISH" if snapshot.close > snapshot.open else "BEARISH" bos = None candle_range = snapshot.high - snapshot.low displacement = "STRONG" if candle_range > 0 and (abs(snapshot.close - snapshot.open) / candle_range) >= 0.7 else "MODERATE" reversal = displacement == "STRONG" and trend == "NEUTRAL" return Structure(swing_high=swing_high, swing_low=swing_low, trend=trend, bos=bos, reversal=reversal, displacement=displacement, atr=atr) def detect_liquidity(self, snapshot, structure): return Liquidity( buy_side=[structure.swing_high] if structure.swing_high else [], sell_side=[structure.swing_low] if structure.swing_low else [] ) def detect_session(self) -> SessionState: utc_hour = datetime.now(timezone.utc).hour if 7 <= utc_hour < 16: name, phase = "LONDON", "ACTIVE" elif 12 <= utc_hour < 21: name, phase = "NY", "ACTIVE" elif 0 <= utc_hour < 7: name, phase = "ASIA", "ACTIVE" else: name, phase = "OFF", "LOW_LIQUIDITY" if 12 <= utc_hour < 16: name = "LONDON_NY_OVERLAP" phase = "HIGH_VOLATILITY" return SessionState(name=name, phase=phase, utc_hour=utc_hour) def detect_location(self, snapshot, structure): if structure.swing_high is None or structure.swing_low is None: return "UNKNOWN" midpoint = (structure.swing_high + structure.swing_low) / 2 if snapshot.close > midpoint: return "UPPER" if snapshot.close < midpoint: return "LOWER" return "MID" def detect_market_cycle(self, structure): if structure.trend == "BULLISH": return "UPSIDE_EXPANSION" elif structure.trend == "BEARISH": return "DOWNSIDE_EXPANSION" return "CONSOLIDATION" def quality_gate(self, picture) -> float: score = 100.0 if picture.structure.trend == "UNKNOWN": score -= 40.0 if picture.structure.trend == "NEUTRAL": score -= 15.0 if picture.structure.atr == 0.0: score -= 20.0 if not picture.liquidity.buy_side: score -= 10.0 if not picture.liquidity.sell_side: score -= 10.0 if picture.session.phase == "LOW_LIQUIDITY": score -= 15.0 if picture.data_quality and picture.data_quality.reliability < 0.7: score -= 30.0 return max(0.0, min(100.0, score)) # ============================================================ # 9. PLAN ENGINE (SL-FIRST & SPREAD ADJUSTED) # ============================================================ class PLANEngine: def build_plans(self, picture: MAPPicture) -> List[Plan]: if not picture.ready: return [] plans = [] spread = picture.snapshot.spread atr = picture.structure.atr if atr == 0: atr = max(1.0, picture.snapshot.high - picture.snapshot.low) if picture.structure.trend == "BULLISH" and picture.structure.swing_low is not None: ref = picture.structure.swing_low entry_low = round(ref + atr * 0.2, 5) entry_high = round(ref + atr * 0.5, 5) tech_sl = round(ref - atr * 0.5, 5) exec_sl = round(tech_sl - spread, 5) risk = entry_low - exec_sl tp1 = round(picture.structure.swing_high, 5) if picture.structure.swing_high else round(entry_low + risk * 1.5, 5) tp2 = round(entry_low + risk * 3.0, 5) rr = (tp1 - entry_low) / risk if risk > 0 else 0.0 plan = Plan( plan_id="PLAN_A", name="PULLBACK TRAP BUY", side="BUY", entry_low=entry_low, entry_high=entry_high, technical_sl=tech_sl, spread=spread, execution_sl=exec_sl, tp1=tp1, tp2=tp2, trigger=f"Price enters zone [{entry_low}-{entry_high}] with structural rejection", invalidation=f"Close below execution SL {exec_sl}", state=TrapState.ARMED, action=Action.WATCH, armed_timestamp=picture.snapshot.timestamp, armed_before_entry=True, atr_at_creation=atr, rr=round(rr, 2) ) plans.append(plan) elif picture.structure.trend == "BEARISH" and picture.structure.swing_high is not None: ref = picture.structure.swing_high entry_low = round(ref - atr * 0.5, 5) entry_high = round(ref - atr * 0.2, 5) tech_sl = round(ref + atr * 0.5, 5) exec_sl = round(tech_sl + spread, 5) risk = exec_sl - entry_high tp1 = round(picture.structure.swing_low, 5) if picture.structure.swing_low else round(entry_high - risk * 1.5, 5) tp2 = round(entry_high - risk * 3.0, 5) rr = (entry_high - tp1) / risk if risk > 0 else 0.0 plan = Plan( plan_id="PLAN_A", name="BREAKOUT FAILURE SHORT", side="SELL", entry_low=entry_low, entry_high=entry_high, technical_sl=tech_sl, spread=spread, execution_sl=exec_sl, tp1=tp1, tp2=tp2, trigger=f"Price reaches zone [{entry_low}-{entry_high}] with structural rejection", invalidation=f"Close above execution SL {exec_sl}", state=TrapState.ARMED, action=Action.WATCH, armed_timestamp=picture.snapshot.timestamp, armed_before_entry=True, atr_at_creation=atr, rr=round(rr, 2) ) plans.append(plan) return plans # ============================================================ # 10. TRAP STATE ENGINE # ============================================================ class TrapEngine: @staticmethod def update(plan: Plan, current_price: float) -> Plan: if plan.side == "BUY" and current_price <= plan.execution_sl: plan.state = TrapState.INVALIDATED plan.action = Action.CANCEL return plan if plan.side == "SELL" and current_price >= plan.execution_sl: plan.state = TrapState.INVALIDATED plan.action = Action.CANCEL return plan if plan.entry_low <= current_price <= plan.entry_high: plan.state = TrapState.TRIGGERED plan.action = Action.EXECUTE return plan plan.state = TrapState.ARMED plan.action = Action.WATCH return plan # ============================================================ # 11. NAVI ENGINE # ============================================================ class NAVIEngine: def route(self, plans: List[Plan]) -> NaviRoute: triggered = [p for p in plans if p.state in (TrapState.TRIGGERED, TrapState.ACTIVE)] armed = [p for p in plans if p.state == TrapState.ARMED] invalidated = [p for p in plans if p.state in (TrapState.INVALIDATED, TrapState.CANCELLED)] primary = triggered[0].plan_id if triggered else (armed[0].plan_id if armed else None) secondary = armed[1].plan_id if len(armed) > 1 else None no_go = [p.plan_id for p in invalidated] if triggered: final_action = Action.MANAGE elif armed: final_action = Action.WATCH else: final_action = Action.NONE return NaviRoute(primary=primary, secondary=secondary, no_go=no_go, final_action=final_action) # ============================================================ # 12. MNS CORE PIPELINE V3.3 # ============================================================ class MNS: def __init__(self): self.map_engine = MAPEngine() self.plan_engine = PLANEngine() self.trap_engine = TrapEngine() self.navi_engine = NAVIEngine() def run(self, snapshot: MarketSnapshot, history: Optional[List[Dict]] = None) -> Dict[str, Any]: data_quality = DataValidator.quality_check(snapshot) picture_preview = self.map_engine.build(snapshot, history) preflight = PreFlight.run(snapshot, picture_preview.event, data_quality) if preflight["status"] == SystemStatus.HALT.value: logger.warning(f"PRE-FLIGHT HALT: {preflight['errors']}") return {"system": "HALT", "stage": "PRE_FLIGHT", "preflight": preflight, "quality": asdict(data_quality)} validation = DataValidator.validate(snapshot) if not validation["valid"]: logger.warning(f"VALIDATION HALT: {validation['errors']}") return {"system": "HALT", "stage": "DATA_VALIDATION", "validation": validation} picture = picture_preview if not picture.ready: logger.warning(f"MAP QUALITY FAIL: Score {picture.quality_score} Conflicts {picture.conflicts}") return {"system": "HALT", "stage": "MAP_QUALITY_GATE", "reason": f"SCORE {picture.quality_score}", "map": asdict(picture)} plans = self.plan_engine.build_plans(picture) for plan in plans: self.trap_engine.update(plan, snapshot.close) route = self.navi_engine.route(plans) return { "system": SystemStatus.READY.value, "preflight": preflight, "map": asdict(picture), "plans": [asdict(p) for p in plans], "navi": asdict(route), "quality": asdict(data_quality) } # ====================
