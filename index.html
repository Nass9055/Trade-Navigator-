MARKET DATA # ============================================================ @dataclass class MarketSnapshot: symbol: str timeframe: str bid: float ask: float timestamp: str open: float high: float low: float close: float source: str = "UNKNOWN" @property def spread(self) -> float: """ Spread belongs to MAP data. PLAN is responsible for using this value. NAVI does not calculate it. """ return round(self.ask - self.bid, 5) # ============================================================ # 3. SYSTEM PRE-FLIGHT SCAN # ============================================================ class PreFlight: REQUIRED_MODULES = ["DATA", "MAP", "PLAN", "NAVI"] @staticmethod def run(snapshot: MarketSnapshot) -> Dict[str, Any]: errors = [] if not snapshot.symbol: errors.append("MISSING_SYMBOL") if not snapshot.timeframe: errors.append("MISSING_TIMEFRAME") if snapshot.bid <= 0: errors.append("INVALID_BID") if snapshot.ask <= 0: errors.append("INVALID_ASK") if snapshot.ask < snapshot.bid: errors.append("ASK_BELOW_BID") if snapshot.high < snapshot.low: errors.append("INVALID_CANDLE_RANGE") if not snapshot.timestamp: errors.append("MISSING_TIMESTAMP") if not snapshot.source: errors.append("MISSING_DATA_SOURCE") spread = snapshot.spread if spread < 0: errors.append("INVALID_SPREAD") if errors: return { "status": SystemStatus.HALT.value, "errors": errors, "spread": spread, "modules": PreFlight.REQUIRED_MODULES, } return { "status": SystemStatus.READY.value, "errors": [], "spread": spread, "modules": PreFlight.REQUIRED_MODULES, } # ============================================================ # 8. DATA VALIDATION # ============================================================ class DataValidator: @staticmethod def validate(snapshot: MarketSnapshot) -> Dict[str, Any]: errors = [] if snapshot.high < snapshot.low: errors.append("HIGH_LOWER_THAN_LOW") if not (snapshot.low <= snapshot.open <= snapshot.high): errors.append("OPEN_OUTSIDE_CANDLE") if not (snapshot.low <= snapshot.close <= snapshot.high): errors.append("CLOSE_OUTSIDE_CANDLE") if snapshot.bid <= 0: errors.append("INVALID_BID") if snapshot.ask <= 0: errors.append("INVALID_ASK") return { "valid": len(errors) == 0, "errors": errors, } # ============================================================ # 9. MAP ENGINE # ============================================================ class MAPEngine: def build(self, snapshot: MarketSnapshot) -> MAPPicture: structure = self.detect_structure(snapshot) liquidity = self.detect_liquidity(snapshot, structure) session = self.detect_session() event = self.detect_event() intelligence = self.collect_intelligence() location = self.detect_location(snapshot, structure) cycle = self.detect_market_cycle(structure) picture = MAPPicture( snapshot=snapshot, structure=structure, liquidity=liquidity, session=session, event=event, intelligence=intelligence, location=location, market_cycle=cycle, ) picture.quality_score = self.quality_gate(picture) picture.ready = picture.quality_score >= 70 return picture def detect_structure(self, snapshot: MarketSnapshot) -> Structure: if snapshot.close > snapshot.open: trend = "BULLISH" elif snapshot.close < snapshot.open: trend = "BEARISH" else: trend = "NEUTRAL" displacement = None candle_range = snapshot.high - snapshot.low if candle_range > 0: body = abs(snapshot.close - snapshot.open) body_ratio = body / candle_range if body_ratio >= 0.70: displacement = "STRONG" elif body_ratio >= 0.40: displacement = "MODERATE" return Structure( swing_high=snapshot.high, swing_low=snapshot.low, trend=trend, displacement=displacement, ) def detect_liquidity(self, snapshot, structure): return Liquidity( buy_side=[structure.swing_high] if structure.swing_high else [], sell_side=[structure.swing_low] if structure.swing_low else [], ) def detect_session(self): return SessionState(name="UNKNOWN", phase="UNKNOWN") def detect_event(self): return EventState(name=None, event_time=None, importance="NONE") def collect_intelligence(self): return MarketIntelligence() def detect_location(self, snapshot, structure): if structure.swing_high is None or structure.swing_low is None: return "UNKNOWN" midpoint = (structure.swing_high + structure.swing_low) / 2 if snapshot.close > midpoint: return "UPPER" if snapshot.close < midpoint: return "LOWER" return "MID" def detect_market_cycle(self, structure): if structure.trend == "BULLISH": if structure.displacement == "STRONG": return "UPSIDE_EXPANSION" return "BULLISH_DEVELOPMENT" if structure.trend == "BEARISH": if structure.displacement == "STRONG": return "DOWNSIDE_EXPANSION" return "BEARISH_DEVELOPMENT" return "UNDEFINED" def quality_gate(self, picture): score = 100 if picture.structure.trend == "UNKNOWN": score -= 25 if not picture.liquidity.buy_side: score -= 10 if not picture.liquidity.sell_side: score -= 10 if picture.conflicts: score -= 30 return max(0, min(100, score)) # ============================================================ # 10. PLAN ENGINE # ============================================================ class PLANEngine: def build_plans(self, picture: MAPPicture) -> List[Plan]: if not picture.ready: return [] plans = [] spread = picture.snapshot.spread trend = picture.structure.trend if trend == "BULLISH": reference = picture.structure.swing_low if reference is not None: entry_low = reference + 2 entry_high = reference + 5 technical_sl = reference - 3 execution_sl = round(technical_sl - spread, 5) tp1 = picture.structure.swing_high tp2 = picture.structure.swing_high + (picture.structure.swing_high - reference) plan = Plan( plan_id="PLAN_A", name="PULLBACK TRAP BUY", side="BUY", entry_low=round(entry_low, 5), entry_high=round(entry_high, 5), technical_sl=round(technical_sl, 5), spread=spread, execution_sl=execution_sl, tp1=round(tp1, 5) if tp1 else 0.0, tp2=round(tp2, 5) if tp2 else 0.0, trigger="Price enters prepared pullback zone and confirms support rejection", invalidation="Price breaks execution SL", state=TrapState.ARMED, action=Action.WATCH, armed_timestamp=picture.snapshot.timestamp, armed_before_entry=True, ) plans.append(plan) elif trend == "BEARISH": reference = picture.structure.swing_high if reference is not None: entry_low = reference - 5 entry_high = reference - 2 technical_sl = reference + 3 execution_sl = round(technical_sl + spread, 5) tp1 = picture.structure.swing_low swing_range = reference - (picture.structure.swing_low if picture.structure.swing_low else reference) tp2 = (picture.structure.swing_low if picture.structure.swing_low else reference) - (swing_range * 0.5) plan = Plan( plan_id="PLAN_A", name="BREAKOUT FAILURE SHORT", side="SELL", entry_low=round(entry_low, 5), entry_high=round(entry_high, 5), technical_sl=round(technical_sl, 5), spread=spread, execution_sl=execution_sl, tp1=round(tp1, 5) if tp1 else 0.0, tp2=round(tp2, 5), trigger="Price reaches prepared breakout-failure zone and confirms rejection", invalidation="Price breaks execution SL", state=TrapState.ARMED, action=Action.WATCH, armed_timestamp=picture.snapshot.timestamp, armed_before_entry=True, ) plans.append(plan) return plans # ============================================================ # 11. TRAP STATE ENGINE # ============================================================ class TrapEngine: @staticmethod def update(plan: Plan, current_price: float, trap_was_armed: bool) -> Plan: if not trap_was_armed: if plan.entry_low <= current_price <= plan.entry_high: plan.state = TrapState.MISSED plan.action = Action.NONE return plan if plan.side == "BUY" and current_price <= plan.execution_sl: plan.state = TrapState.INVALIDATED plan.action = Action.CANCEL return plan if plan.side == "SELL" and current_price >= plan.execution_sl: plan.state = TrapState.INVALIDATED plan.action = Action.CANCEL return plan if plan.entry_low <= current_price <= plan.entry_high: plan.state = TrapState.TRIGGERED plan.action = Action.EXECUTE return plan plan.state = TrapState.ARMED plan.action = Action.WATCH return plan # ============================================================ # 12. NAVI ENGINE # ============================================================ class NAVIEngine: def route(self, plans: List[Plan]) -> NaviRoute: triggered = [p for p in plans if p.state in (TrapState.TRIGGERED, TrapState.ACTIVE)] armed = [p for p in plans if p.state == TrapState.ARMED] invalidated = [p for p in plans if p.state in (TrapState.INVALIDATED, TrapState.CANCELLED)] missed = [p for p in plans if p.state == TrapState.MISSED] primary = triggered[0].plan_id if triggered else (armed[0].plan_id if armed else None) candidates = [p for p in armed if p.plan_id != primary] secondary = candidates[0].plan_id if candidates else None no_go = [plan.plan_id for plan in invalidated] + [plan.plan_id + ":MISSED" for plan in missed] if triggered: final_action = Action.MANAGE elif armed: final_action = Action.WATCH else: final_action = Action.NONE return NaviRoute(primary=primary, secondary=secondary, no_go=no_go, final_action=final_action) # ============================================================ # 13.
